// 期間內部代號（python 端 label）→ 面板顯示用的中文措辭。
var PERIOD_ZH = {'上次財報前後':'上次財報以來', '近1個月':'近一個月', '近3個月':'近三個月'};

function pctWord(v, upWord, downWord){
  if(v==null||Number.isNaN(v)) return null;
  return (v>=0?upWord:downWord)+' '+Math.abs(v).toFixed(2)+'%';
}

function renderRevVsPrice(vm){
  var c = vm.c2;
  var tblEl = document.getElementById('revVsPriceTable'), sentEl = document.getElementById('revVsPriceSentence');
  if(c.status!=='ok'){ noData(tblEl, c.reason); if(sentEl) sentEl.innerHTML=''; return; }
  var periods = c.periods || [];
  tblEl.innerHTML =
    '<thead><tr><th>期間</th><th class="num">明年度 EPS<br>預估變動</th><th class="num">股價漲跌</th><th class="num">隱含本益比<br>變動</th></tr></thead><tbody>'+
    periods.map(function(p){
      var lbl = PERIOD_ZH[p.label] || p.label;
      return '<tr><td>'+lbl+'<span class="base-date">基準：'+(p.base_date||'無資料')+'</span></td>'+
        '<td class="num">'+fmtPctColored(p.eps_fy_next_chg_pct)+'</td>'+
        '<td class="num">'+fmtPctColored(p.price_chg_pct)+'</td>'+
        '<td class="num">'+fmtPctParen(p.implied_ntm_pe_chg_pct)+'</td></tr>';
    }).join('')+'</tbody>';

  // 機械句：只用上表數字，選基準日最早（涵蓋期間最長）且三個數字都有資料的一格。
  var complete = periods.filter(function(p){
    return p.base_date && p.eps_fy_next_chg_pct!=null && p.price_chg_pct!=null && p.implied_ntm_pe_chg_pct!=null;
  }).sort(function(a,b){ return a.base_date<b.base_date?-1:(a.base_date>b.base_date?1:0); });
  if(!sentEl) return;
  if(!complete.length){ sentEl.innerHTML = '無資料：三個期間都缺基準日或估值資料'; return; }
  var p = complete[0], lbl = PERIOD_ZH[p.label] || p.label;
  sentEl.innerHTML = lbl+'明年度 EPS 預估'+pctWord(p.eps_fy_next_chg_pct,'上修','下修')+
    '，股價'+pctWord(p.price_chg_pct,'漲','跌')+
    '，隱含本益比'+pctWord(p.implied_ntm_pe_chg_pct,'上升','下降')+'。';
}
