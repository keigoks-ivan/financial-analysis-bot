// 期間內部代號（python 端 label）→ 面板顯示用的中文措辭。
var PERIOD_ZH = {'上次財報前後':'上次財報以來', '近1個月':'近一個月', '近3個月':'近三個月'};

function pctWord(v, upWord, downWord){
  if(v==null||Number.isNaN(v)) return null;
  return (v>=0?upWord:downWord)+' '+Math.abs(v).toFixed(2)+'%';
}

function renderRevVsPrice(vm){
  var c = vm.c2;
  var tblEl = document.getElementById('revVsPriceTable'), sentEl = document.getElementById('revVsPriceSentence');
  if(c.status!=='ok'){ noData(tblEl, c.reason); if(sentEl) sentEl.innerHTML=''; return; }
  var periods = c.periods || [];
  // 2026-10-08 前建置的逐檔資料沒有日曆年欄位：沿用明年度欄位顯示，等每日重建後自動換成日曆年
  var legacy = periods.length>0 && periods[0].eps_cy_chg_pct===undefined && periods[0].eps_fy_next_chg_pct!==undefined;
  if(legacy) periods.forEach(function(p){ p.eps_cy_chg_pct = p.eps_fy_next_chg_pct; });
  var cyYear = legacy ? '明年度' : (periods[0]&&periods[0].eps_cy_year)? periods[0].eps_cy_year+' 年' : '篩選年度';
  var cyReason = (periods[0]&&periods[0].eps_cy_reason)||null;
  tblEl.innerHTML =
    '<thead><tr><th>期間</th><th class="num">'+cyYear+' EPS<br>預估變動</th><th class="num">股價漲跌</th><th class="num">隱含本益比<br>變動</th></tr></thead><tbody>'+
    periods.map(function(p){
      var lbl = PERIOD_ZH[p.label] || p.label;
      return '<tr><td>'+lbl+'<span class="base-date">基準：'+(p.base_date||'無資料')+'</span></td>'+
        '<td class="num">'+(p.eps_cy_chg_pct==null&&cyReason?'無資料<span class="base-date">'+cyReason+'</span>':fmtPctColored(p.eps_cy_chg_pct))+'</td>'+
        '<td class="num">'+fmtPctColored(p.price_chg_pct)+'</td>'+
        '<td class="num">'+fmtPctParen(p.implied_ntm_pe_chg_pct)+'</td></tr>';
    }).join('')+'</tbody>';

  // 機械句：只用上表數字，選基準日最早（涵蓋期間最長）且三個數字都有資料的一格。
  var complete = periods.filter(function(p){
    return p.base_date && p.eps_cy_chg_pct!=null && p.price_chg_pct!=null && p.implied_ntm_pe_chg_pct!=null;
  }).sort(function(a,b){ return a.base_date<b.base_date?-1:(a.base_date>b.base_date?1:0); });
  if(!sentEl) return;
  if(!complete.length){ sentEl.innerHTML = '無資料：三個期間都缺基準日或估值資料'; return; }
  var p = complete[0], lbl = PERIOD_ZH[p.label] || p.label;
  sentEl.innerHTML = lbl+' '+cyYear+' EPS 預估'+pctWord(p.eps_cy_chg_pct,'上修','下修')+
    '，股價'+pctWord(p.price_chg_pct,'漲','跌')+
    '，隱含本益比'+pctWord(p.implied_ntm_pe_chg_pct,'上升','下降')+'。';
}

// 2026-09-29：個股研究總覽（原 /t/{T}.html）併入本頁「本站研究」章節——資料獨立
// fetch docs/t/data/{T}.json（build_ticker_hubs.py 輸出，與該 ticker hub 頁同源），
// 找不到就顯示一行「這檔還沒有本站研究。」不擋主頁其餘章節渲染。
var CKPT_STATUS_ZH = {ok:'正常', warning:'警示', triggered:'已觸發', unknown:'未知'};
function renderResearchChapter(container, data){
  if(!container) return;
  if(!data){ container.innerHTML = '<div class="nodata">這檔還沒有本站研究。</div>'; return; }

  var parts = [];
  var ckpt = data.checkpoint || {};
  var km = ckpt.kill_metrics || [], cats = ckpt.catalysts || [], rearm = ckpt.rearm_trigger || '';
  if(km.length){
    parts.push('<div class="exhibit"><div class="ex-title">證偽觸發器</div>'+
      '<div class="tbl-wrap"><table class="imq-table"><thead><tr><th>指標</th><th>空頭門檻</th><th>窗口</th><th>狀態</th></tr></thead><tbody>'+
      km.map(function(k){
        return '<tr><td>'+moatEsc(k.metric||'—')+'</td><td>'+moatEsc(k.bear_threshold||'—')+'</td>'+
          '<td>'+moatEsc(k.window||'—')+'</td><td>'+moatEsc(CKPT_STATUS_ZH[k.status]||k.status||'—')+'</td></tr>';
      }).join('')+'</tbody></table></div></div>');
  }
  if(cats.length){
    var today = new Date().toISOString().slice(0,10);
    var sortedCats = cats.slice().sort(function(a,b){ return (a.date||'') < (b.date||'') ? -1 : 1; });
    parts.push('<div class="exhibit"><div class="ex-title">催化劑</div>'+
      '<div class="tbl-wrap"><table class="imq-table"><thead><tr><th>日期</th><th>事件</th><th>影響</th></tr></thead><tbody>'+
      sortedCats.map(function(c){
        var past = c.date && c.date < today;
        return '<tr'+(past?' style="opacity:.55"':'')+'><td>'+moatEsc(c.date||'—')+
          (past?'（已過）':'')+'</td><td>'+moatEsc(c.event||'—')+'</td><td>'+moatEsc(c.impact||'')+'</td></tr>';
      }).join('')+'</tbody></table></div></div>');
  }
  if(rearm){
    parts.push('<div class="exhibit"><div class="ex-title">再武裝條件</div>'+
      '<p style="font-size:12.5px;color:var(--body)">'+moatEsc(rearm)+'</p></div>');
  }

  var lists = data.lists || {};
  function listBlock(title, rows, cols, rowFn){
    if(!rows || !rows.length) return '';
    return '<div class="exhibit"><div class="ex-title">'+title+'（'+rows.length+' 筆）</div>'+
      '<div class="tbl-wrap"><table class="imq-table"><thead><tr>'+cols.map(function(c){return '<th>'+c+'</th>';}).join('')+'</tr></thead><tbody>'+
      rows.map(rowFn).join('')+'</tbody></table></div></div>';
  }
  parts.push(listBlock('個股研究', lists.dd, ['日期','報告','裁決'], function(r){
    return '<tr><td>'+moatEsc(r.date||'—')+'</td><td><a href="'+moatEsc(r.path)+'">'+
      (r.brief?'DD 快速版':'DD 報告')+'</a>'+(r.era?'　'+moatEsc(r.era):'')+'</td><td>'+
      moatEsc(r.verdict&&r.verdict!=='—'?r.verdict:'—')+'</td></tr>';
  }));
  parts.push(listBlock('所屬產業研究', lists.id, ['日期','主題','供需 / 時鐘'], function(r){
    var sub = [];
    if(r.sd_verdict) sub.push('供需 '+r.sd_verdict);
    if(r.clock_phase) sub.push('時鐘 '+r.clock_phase);
    return '<tr><td>'+moatEsc(r.date||'—')+'</td><td><a href="'+moatEsc(r.path)+'">'+moatEsc(r.theme)+'</a></td><td>'+moatEsc(sub.join(' · '))+'</td></tr>';
  }));
  parts.push(listBlock('供應鏈位置', lists.supply_chain, ['主題','節點'], function(r){
    return '<tr><td><a href="'+moatEsc(r.path)+'">'+moatEsc(r.title)+'</a></td><td>'+moatEsc((r.nodes||[]).slice(0,6).join('、'))+'</td></tr>';
  }));
  parts.push(listBlock('對比研究', lists.comparisons, ['日期','對比'], function(r){
    var others = (r.tickers||[]).filter(function(t){ return t!==data.ticker; }).join('、');
    return '<tr><td>'+moatEsc(r.date||'—')+'</td><td><a href="'+moatEsc(r.path)+'">多股對比'+(others?'（vs '+moatEsc(others)+'）':'')+'</a></td></tr>';
  }));
  parts.push(listBlock('期望落差綜合', lists.synthesis, ['日期','報告'], function(r){
    return '<tr><td>'+moatEsc(r.date||'—')+'</td><td><a href="'+moatEsc(r.path)+'">期望落差綜合研判</a></td></tr>';
  }));

  var html = parts.join('');
  if(!html){ container.innerHTML = '<div class="nodata">這檔還沒有本站研究。</div>'; return; }
  container.innerHTML = html + '<div class="ex-src">資料來源：/t/data/'+moatEsc(data.ticker)+'.json（個股 DD／所屬產業 ID／供應鏈位置／對比研究／期望落差綜合研判聚合）</div>';
}
